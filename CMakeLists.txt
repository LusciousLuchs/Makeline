cmake_minimum_required(VERSION 3.16)
# ---- version: the one place to change it. See CHANGELOG.md. Pushing a new version to main publishes a release.
project(Makeline VERSION 0.3.1 LANGUAGES C CXX)
set(MAKELINE_STAGE "alpha" CACHE STRING "Release stage: dev, alpha, beta, rc, or empty for a release")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(TP ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
set(RL ${TP}/raylib/src)

# The game ships for Windows. On Linux this builds the rules (src/core) and their tests;
# MAKELINE_NATIVE_APP=ON also builds the game natively (X11), for automated screenshots.
if(WIN32)
  set(MAKELINE_BUILD_APP ON)
else()
  option(MAKELINE_NATIVE_APP "Developer only: also build the game natively (X11)" OFF)
  set(MAKELINE_BUILD_APP ${MAKELINE_NATIVE_APP})
endif()

set(MAKELINE_VERSION ${PROJECT_VERSION})
if(MAKELINE_STAGE)
  set(MAKELINE_VERSION ${MAKELINE_VERSION}-${MAKELINE_STAGE})
endif()
configure_file(src/core/version.hpp.in ${CMAKE_CURRENT_BINARY_DIR}/generated/version.hpp @ONLY)
message(STATUS "Makeline ${MAKELINE_VERSION}")

find_package(Threads REQUIRED)
file(GLOB CORE_SRC CONFIGURE_DEPENDS src/core/*.cpp)
add_library(makeline_core STATIC ${CORE_SRC})
target_include_directories(makeline_core PUBLIC src/core ${TP}/json ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(makeline_core PUBLIC Threads::Threads)

if(MAKELINE_BUILD_APP)
  add_library(raylib STATIC
    ${RL}/rcore.c ${RL}/rshapes.c ${RL}/rtextures.c ${RL}/rtext.c
    ${RL}/rmodels.c ${RL}/utils.c ${RL}/raudio.c ${RL}/rglfw.c)
  target_include_directories(raylib PUBLIC ${RL} PRIVATE ${RL}/external/glfw/include)
  target_compile_definitions(raylib PUBLIC PLATFORM_DESKTOP GRAPHICS_API_OPENGL_33)
  target_compile_definitions(raylib PRIVATE SUPPORT_FILEFORMAT_JPG=1)
  if(WIN32)
    target_link_libraries(raylib PUBLIC opengl32 gdi32 winmm)
  else()
    target_compile_definitions(raylib PRIVATE _GLFW_X11)
    target_compile_options(raylib PRIVATE "SHELL:-idirafter ${TP}/x11-fallback")
    target_link_libraries(raylib PUBLIC m dl pthread)
  endif()

  file(GLOB GAME_SRC CONFIGURE_DEPENDS src/game/*.cpp)
  set(EXTRA_SRC "")
  if(WIN32)   # version and studio shown in the .exe's Properties > Details
    enable_language(RC)
    set(MAKELINE_RC_FLAGS 0x0L)
    if(MAKELINE_STAGE)
      set(MAKELINE_RC_FLAGS 0x2L)
    endif()
    configure_file(src/makeline.rc.in ${CMAKE_CURRENT_BINARY_DIR}/generated/makeline.rc @ONLY)
    set(EXTRA_SRC ${CMAKE_CURRENT_BINARY_DIR}/generated/makeline.rc)
  endif()
  add_executable(makeline ${GAME_SRC} ${EXTRA_SRC})
  target_include_directories(makeline PRIVATE src)
  target_link_libraries(makeline PRIVATE makeline_core raylib)
  if(WIN32)
    set_target_properties(makeline PROPERTIES WIN32_EXECUTABLE ON)
    target_link_libraries(makeline PRIVATE winhttp)
    if(MSVC)
      target_link_options(makeline PRIVATE /ENTRY:mainCRTStartup)
    endif()
  endif()
  if(NOT MSVC)
    target_link_options(makeline PRIVATE -static-libstdc++ -static-libgcc)
  endif()
endif()

add_executable(makeline_tests tests/tests.cpp)
target_link_libraries(makeline_tests PRIVATE makeline_core)
enable_testing()
add_test(NAME rules COMMAND makeline_tests)
